#   1. Перенесіть сюди зі свого старого .env рядки з ключами (*_API_KEY, *_SECRET, *_PASSWORD),
#      TELEGRAM_BOT_TOKEN, TELEGRAM_CHAT_ID, TELEGRAM_PIN, KEYS_SECRET (якщо були).
#   2. Перейменуйте старий .env на .env.old, а цей файл — на .env
#   3. Видаліть data\settings.json (там зміни, зроблені через Telegram, — вони важливіші за .env).
#      Ключі, додані через Telegram, лежать окремо в data\keys.json — їх НЕ видаляйте.
# Змінено відносно стандартних значень:
#   EXCHANGES          — лише ваші 4 біржі з ключами
#   MAX_TRADE_USDT=15, LEVERAGE=2, MAX_POSITIONS=1, MIN_TRADE_USDT=10 — малі суми для першого live
#   MIN_FUNDING_PCT=0.3, MIN_PROFIT_PCT=0.2 — вхід лише коли після комісій (реальних, 0.1% на Bybit/Bitget)
#                        і несприятливої різниці цін лишається ≥ 0.2%
#   MAX_ENTRY_BASIS_PCT=0.5 — дозволено заходити з різницею цін проти нас до 0.5%, якщо фандінг її покриває
#   EXIT_AFTER_SEC=5, SETTLE_DELAY_SEC=5 — пошук виходу починається через 5 с після виплати
#   MAX_CONVERGE_WAIT_MIN=30 — якщо вихід у мінус, чекати до 30 хв, потім вийти в будь-якому разі
#   MAX_DAILY_LOSS_USDT=5 — автопауза при збитку за день > 5 USDT
#   MAX_WINDOW_HOURS=0 — кожна угода лише під одну виплату, без утримання годинами
# ============================================================================

# --- Розмір і ризик ---
# auto  — бот дивиться стакани обох бірж (вхід і вихід), вільну маржу і заходить на максимально можливу суму,
#         але не більше MAX_TRADE_USDT. Розмір зменшується, якщо великий обсяг погіршує ціну/профіт.
# coins — вхід і вихід фіксованою КІЛЬКІСТЮ МОНЕТ (COIN_QTY), usdt — фіксованою сумою POSITION_USDT
SIZE_MODE=auto
MAX_TRADE_USDT=15          # ВАША СУМА (маржа) на 1 угоду на кожній біржі; позиція = сума × плече (якщо стакани дозволяють)
MIN_TRADE_USDT=10          # якщо стакани дозволяють позицію меншу за це — не заходити
CHECK_EXIT_DEPTH=true      # враховувати і стакан на вихід
COIN_QTY=                  # BTC:0.002,ETH:0.05 — у coins: точна кількість; в auto: макс. кількість цієї монети
POSITION_USDT=100          # номінал КОЖНОЇ ноги (для SIZE_MODE=usdt)
LEVERAGE=2                 # бажане плече; фактичне однакове на обох біржах і не більше максимуму кожної біржі для монети
MARGIN_MODE=isolated
MAX_POSITIONS=1

# --- Вхід (під конкретну виплату фандінгу) ---
MIN_FUNDING_PCT=0.3        # ВХІД лише якщо фандінг за виплату(и) більший за цей %. Напр. OKX 3% vs Binance 1% = 2%
MIN_PROFIT_PCT=0.2         # і чистий профіт після 4 тейкер-комісій більший за цей %
ENTRY_BEFORE_SEC=120       # заходити приблизно за 2 хв до виплати
ENTRY_MIN_SEC=20           # якщо до виплати менше — вже не заходимо
MAX_WINDOW_HOURS=0         # 0 = кожна угода під ОДНУ конкретну виплату: вхід ~2 хв до неї, вихід одразу після
ALLOW_ESTIMATED=true       # враховувати проміжні виплати швидшої біржі (1h) за поточною ставкою
LOOKAHEAD_HOURS=8          # горизонт для /top
MAX_ENTRY_BASIS_PCT=0.5    # макс. несприятлива різниця цін між біржами на вході
BASIS_COST=true            # true: несприятлива різниця цін на вході віднімається з прибутку; false: не віднімати (див. README)
MAX_SLIPPAGE_PCT=0.1       # макс. відхилення середньої ціни виконання від найкращої в стакані (на кожній нозі)
MAX_ABS_FUNDING_RATE_PCT=10  # ставки більші за це — вважаємо аномалією
DEFAULT_TAKER_FEE_PCT=0.05

# --- Вихід ---
EXIT_AFTER_SEC=5           # через 5 с після (останньої) виплати бот починає шукати точку виходу (перевірка кожні 5 с)
CONVERGE_PCT=0.05          # ...або одразу, як різниця цін між біржами ≤ 0.05% («зійшлися»)
EARLY_EXIT=true            # вийти ще ДО виплати, якщо ціни зійшлися і прибуток від цін уже більший за фандінг
SETTLE_DELAY_SEC=5         # не закриватися раніше ніж через 5 с після виплати (live ще й чекає підтвердження нарахування)
MAX_CONVERGE_WAIT_MIN=30   # якщо після виплати вихід у мінус — чекати сходження цін до N хв (але не до наступної виплати)

# --- Захист коштів ---
ORDER_TYPE=auto            # auto: бот сам — лімітний IOC з межею ціни (звичайні входи/виходи) або ринковий (термінові); market: завжди ринкові
LIQ_CLOSE_PCT=60           # бот закриває пару, коли ціна пройшла 60% шляху від входу до ліквідації будь-якої ноги
LIQ_CHECK_SEC=5            # перевірка ризику ліквідації кожні 5 с (обидві біржі одночасно)
PROTECT_STOPS=true         # стоп-ордери на самих біржах — спрацюють, навіть якщо бот/сервер упав
STOP_AT_PCT=80             # стоп на біржі на 80% шляху до ліквідації (має бути більше за LIQ_CLOSE_PCT)
RECONCILE_MIN=2            # звірка позицій бота з біржами кожні 2 хв (і при кожному старті)
MAX_DAILY_LOSS_USDT=5      # збиток за день більший — автопауза входів (0 = вимк.)
MAX_LOSS_STREAK=3          # стільки збиткових угод поспіль — автопауза (0 = вимк.)
MAX_ERRORS=5               # стільки помилок ордерів за 10 хв — автопауза (0 = вимк.)
MAX_TOTAL_USDT=0           # макс. сумарна позиція всіх угод (одна нога), USDT (0 = без ліміту)
MAX_EXCHANGE_USDT=0        # макс. сумарна позиція на одній біржі, USDT (0 = без ліміту)
RESERVE_PCT=20             # частка вільного балансу, яку бот ніколи не використовує
MIN_VOLUME_USDT=1000000    # мін. обсяг торгів монети за 24 год на кожній біржі пари
NEW_LISTING_DAYS=3         # не торгувати монети, лістинговані менше N днів тому
DELIST_DAYS=14             # не торгувати монети з делістингом/експірацією в найближчі N днів
MAX_CLOCK_SKEW_MS=1500     # попередження, якщо годинник сервера розходиться з біржами
HEARTBEAT_HOURS=3          # «💓 я працюю» в Telegram кожні N год (0 = вимк.)
SELFTEST_ON_START=true     # live: самоперевірка кожної біржі з ключем при старті; не пройшла — бот на ній не торгує
FUNDING_CONFIRM_SEC=90     # live: не виходити, доки біржа не покаже нарахування фандінгу (максимум N с)
AUTOBAN_LOSSES=3           # монета дала N збиткових угод поспіль — у чорний список (0 = вимк.)
IMBALANCE_RATIO=2          # попередити, якщо на одній біржі в N разів більше грошей, ніж на іншій (0 = вимк.)
LOCK_FILE=data/bot.lock    # захист від запуску двох копій бота
LOG_TO_FILE=true           # журнал у файл logs/bot-РРРР-ММ-ДД.log (без ключів — можна надсилати розробнику)
